#include <QtCore/QByteArray>
#include <QtCore/QVector>
#include <QtCore/QString>
#include <QtEndian>
#include <cstring>

struct WavelengthRecord
{
    quint8 channel;
    quint8 sensor;
    float wavelength;
};

struct WavelengthFrame
{
    quint8 version;
    quint8 dataType;
    quint16 recordCount;
    quint32 sequence;
    QVector<WavelengthRecord> records;
};

static bool parseWavelengthFrame(const QByteArray& datagram,
                                 WavelengthFrame& frame,
                                 QString* error = nullptr)
{
    const int headerLen = 10;
    const int recordLen = 6;

    auto fail = [&](const QString& message) {
        if (error)
            *error = message;
        return false;
    };

    if (datagram.size() < headerLen)
        return fail(QStringLiteral("数据包长度小于帧头"));

    const uchar* data =
        reinterpret_cast<const uchar*>(datagram.constData());

    quint16 magic = qFromLittleEndian<quint16>(data + 0);

    if (magic != 0xA55A)
        return fail(QStringLiteral("帧头错误"));

    frame.version = data[2];
    frame.dataType = data[3];

    frame.recordCount =
        qFromLittleEndian<quint16>(data + 4);

    frame.sequence =
        qFromLittleEndian<quint32>(data + 6);

    int expectedLength =
        headerLen + frame.recordCount * recordLen;

    if (datagram.size() != expectedLength)
    {
        return fail(QStringLiteral(
            "长度错误，实际=%1，期望=%2")
            .arg(datagram.size())
            .arg(expectedLength));
    }

    frame.records.clear();
    frame.records.reserve(frame.recordCount);

    int offset = headerLen;

    for (quint16 i = 0; i < frame.recordCount; ++i)
    {
        WavelengthRecord record;

        record.channel = data[offset];
        record.sensor = data[offset + 1];

        quint32 floatBits =
            qFromLittleEndian<quint32>(data + offset + 2);

        std::memcpy(&record.wavelength,
                    &floatBits,
                    sizeof(record.wavelength));

        frame.records.append(record);
        offset += recordLen;
    }

    return true;
}
